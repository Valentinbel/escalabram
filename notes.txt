    @Override
    public EmailMessage.EmailType getSupportedType() {
        return EmailMessage.EmailType.INVOICE; // Ajouté dans l'enum
    }

    @Override
    public EmailContent buildContent(Map<String, Object> context) {
        return new EmailContent(
            "Votre facture #" + context.get("invoiceId"),
            "..."
        );
    }
}
